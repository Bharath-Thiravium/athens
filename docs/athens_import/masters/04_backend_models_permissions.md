## Permission Classes

### IsMasterAdmin Permission
**File**: `/app/backend/authentication/permissions.py`

```python
class IsMasterAdmin(permissions.BasePermission):
    """
    Allows access only to master admin users.
    """
    def has_permission(self, request, view):
        return bool(
            request.user and
            request.user.is_authenticated and
            is_master_user(request.user)
        )

# Helper function
def is_master_user(user):
    """Check if user is a master admin"""
    return (
        user.user_type in ['master', 'masteradmin'] or 
        user.admin_type in ['master', 'masteradmin']
    )
```

## Permission Decorators

### Function-Based View Decorators
```python
def require_master_admin(view_func):
    """Decorator for function-based views that require master admin."""
    def _wrapped_view(request, *args, **kwargs):
        if not (
            request.user and
            request.user.is_authenticated and
            is_master_user(request.user)
        ):
            return permissions.PermissionDenied("You do not have master admin privileges.")
        return view_func(request, *args, **kwargs)
    return _wrapped_view

# Usage
@require_master_admin
def some_master_only_view(request):
    # Master admin only logic
    pass
```

### Class-Based View Decorators
```python
# Applied to entire view class
@permission_classes([IsMasterAdmin])
class MasterAdminProjectCreateView(ScopedWriteMixin, APIView):
    def post(self, request):
        # Master admin only logic
        pass

# Applied to specific methods
class SomeView(APIView):
    @method_decorator(require_master_admin)
    def post(self, request):
        # Master admin only logic
        pass
```

## Tenant & Project Context Enforcement

### ScopedWriteMixin
**File**: `/app/backend/authentication/tenant_scoped_utils.py`

```python
class ScopedWriteMixin:
    """
    Mixin that automatically enforces tenant and project scoping for write operations.
    """
    def perform_create(self, serializer):
        tenant_id = getattr(self.request, "scope_tenant_id", None) or \
                   getattr(self.request.user, "athens_tenant_id", None)
        save_kwargs = {}
        if tenant_id:
            save_kwargs["athens_tenant_id"] = tenant_id
        serializer.save(**save_kwargs)

def enforce_scope_or_403(request):
    """Ensure request has valid tenant scope"""
    if not hasattr(request, 'scope_tenant_id') and not getattr(request.user, 'athens_tenant_id', None):
        raise PermissionDenied("Missing tenant context")

def enforce_object_scope_or_403(request, obj, tenant_attr="athens_tenant_id", project_attr=None):
    """Ensure object belongs to user's tenant/project scope"""
    user_tenant_id = getattr(request.user, "athens_tenant_id", None)
    obj_tenant_id = getattr(obj, tenant_attr, None)
    
    if user_tenant_id and obj_tenant_id and str(user_tenant_id) != str(obj_tenant_id):
        raise PermissionDenied("Access denied: Object belongs to different tenant")
```

## Middleware Integration

### AthensTenantMiddleware
**File**: `/app/backend/authentication/tenant_middleware.py`

```python
class AthensTenantMiddleware(MiddlewareMixin):
    """
    Enforces tenant isolation by validating athens_tenant_id from user context.
    """
    def process_request(self, request):
        # Extract tenant from authenticated user
        if hasattr(request, 'user') and request.user.is_authenticated:
            tenant_id = getattr(request.user, 'athens_tenant_id', None)
            if tenant_id:
                # Validate tenant exists and is active
                tenant = TenantResolver.get_tenant(tenant_id)
                if tenant:
                    request.athens_tenant_id = tenant_id
                    request.athens_tenant = tenant
                else:
                    return JsonResponse({'error': 'Invalid tenant'}, status=403)
```

### ProjectIsolationMiddleware
**File**: `/app/backend/authentication/middleware.py`

```python
class ProjectIsolationMiddleware:
    """
    Enforces project-level isolation for non-master users.
    Masters operate at tenant level and bypass project isolation.
    """
    def __call__(self, request):
        if hasattr(request, 'user') and request.user.is_authenticated:
            # Masters don't have project isolation - they see all projects in tenant
            if not is_master_user(request.user):
                request.user_project = getattr(request.user, 'project', None)
        
        response = self.get_response(request)
        return response
```

## Query Filtering & Isolation

### Tenant-Scoped Queries
```python
# Masters see all projects in their tenant
def get_master_projects(user):
    if user.athens_tenant_id:
        return Project.objects.filter(athens_tenant_id=user.athens_tenant_id)
    else:
        return Project.objects.all()  # Fallback for legacy masters

# Project admins see only their assigned project
def get_user_projects(user):
    if user.project:
        return Project.objects.filter(id=user.project.id)
    else:
        return Project.objects.none()
```

### Automatic Filtering in Views
```python
class ProjectListView(APIView):
    def get(self, request):
        user = request.user
        
        # MASTER ADMIN: Can see all projects in their tenant
        if is_master_user(user):
            if user.athens_tenant_id:
                projects = Project.objects.filter(athens_tenant_id=user.athens_tenant_id)
            else:
                projects = Project.objects.all()
        elif user.project:
            # Regular users see only their assigned project
            projects = Project.objects.filter(id=user.project.id)
        else:
            projects = Project.objects.none()
        
        serializer = ProjectSerializer(projects, many=True)
        return Response(serializer.data)
```

## Model Relationships

### User Hierarchy
```
Master Admin (created via management command or superadmin)
├── Creates Project Admins (client, epc, contractor)
│   └── Project Admins create Admin Users (clientuser, epcuser, contractoruser)
└── Approves AdminDetail and UserDetail submissions
```

### Tenant Relationships
```
athens_tenant_id (UUID)
├── Master Admin (athens_tenant_id = tenant)
├── Projects (athens_tenant_id = tenant)
├── Project Admins (athens_tenant_id = tenant, project = specific project)
└── Admin Users (athens_tenant_id = tenant, project = specific project)
```

### Foreign Key Relationships
```python
# CustomUser relationships
user.project -> Project (null for masters)
user.created_by -> CustomUser (hierarchy tracking)
user.admin_detail -> AdminDetail (one-to-one)
user.company_detail -> CompanyDetail (one-to-one, masters only)

# Project relationships  
project.admins -> CustomUser (reverse FK, all users assigned to project)
project.athens_tenant_id -> Tenant (logical relationship)
```
